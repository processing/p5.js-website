description: >
  <p>Draws <code>count</code> instances of the next 3D primitive or model using
  WebGL

  instanced rendering.</p>

  <p>Call a draw method on the returned object to render that primitive

  <code>count</code> times in a single draw call. Instance-specific transforms
  and

  attributes (position, color, etc.) are supplied through a custom shader

  that reads per-instance data from an instanced attribute buffer.</p>
line: 2699
isConstructor: false
itemtype: method
example:
  - |-
    <div>
    <code>
    let myShader;
    let count = 5;

    function setup() {
      createCanvas(200, 200, WEBGL);
      myShader = buildMaterialShader(drawSpaced);
      describe('Five red spheres arranged in a horizontal line.');
    }

    function drawSpaced() {
      worldInputs.begin();
      // Spread spheres evenly across the canvas based on their index
      let spacing = width / count;
      worldInputs.position.x +=
        (instanceIndex - (count - 1) / 2) * spacing;
      worldInputs.end();
    }

    function draw() {
      background(220);
      lights();
      noStroke();
      fill('red');
      shader(myShader);
      instances(count).sphere(15);
    }
    </code>
    </div>
