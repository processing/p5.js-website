description: >
  <p>Sets a grayscale image that adds bumps and dents to a shape's surface.</p>

  <p>A bump map is a height map: the brightness of the image at each point is

  read as how high the surface is there, and p5.js works out which way the

  surface tilts from how quickly that height changes. Bright areas rise and

  dark areas sink, so lights react to detail that isn't in the geometry.</p>

  <p><code>bumpTexture()</code> works like <a
  href="/reference/p5/texture/"><code>texture()</code></a>, but sets

  the bump map instead of the base color. The parameter, <code>tex</code>, is
  the image

  to use. Passing <code>null</code> clears it, as in
  <code>bumpTexture(null)</code>. The optional

  second parameter, <code>scale</code>, tunes how pronounced the bumps are. The
  map can

  also be scoped between <a href="/reference/p5/push/"><code>push()</code></a>
  and

  <a href="/reference/p5/pop/"><code>pop()</code></a>.</p>

  <p><a href="/reference/p5/normalTexture/"><code>normalTexture()</code></a>
  creates a similar effect

  from a tangent-space normal map, the blue-tinted kind that stores

  directions rather than height. A bump map is usually easier to make by

  hand, since it's just a grayscale picture of where the surface is high and

  low. Only one can be active at a time, so setting one replaces the other.</p>

  <p>A light source is needed to see the effect.</p>

  <p>Note: On a shape whose texture coordinates wrap all the way around, such as

  <a href="/reference/p5/sphere/"><code>sphere()</code></a>, the two edges of
  the image meet. The

  image has to tile for them to line up. Because a bump map is read by

  comparing neighbouring pixels, also call

  <a href="/reference/p5/textureWrap/"><code>textureWrap(REPEAT)</code></a> so
  those comparisons

  carry across the join instead of stopping at the edge.</p>

  <p>Note: <code>bumpTexture()</code> can only be used in WebGL mode.</p>

example:
  - |-
    // Click and drag the mouse to view the scene from different angles.

    let bumpMap;

    function setup() {
      createCanvas(100, 100, WEBGL);

      // Build a grayscale height map with a grid of round bumps.
      bumpMap = createImage(64, 64);
      bumpMap.loadPixels();
      for (let y = 0; y < bumpMap.height; y += 1) {
        for (let x = 0; x < bumpMap.width; x += 1) {
          // Bright where the surface is high, dark where it's low. The pattern
          // repeats a whole number of times across the image so it tiles,
          // which keeps it from showing a seam where the sphere's texture
          // coordinates wrap around.
          let h = sin((x / bumpMap.width) * TWO_PI * 4);
          h *= sin((y / bumpMap.height) * TWO_PI * 4);
          let v = (h * 0.5 + 0.5) * 255;
          let i = (x + y * bumpMap.width) * 4;
          bumpMap.pixels[i] = v;
          bumpMap.pixels[i + 1] = v;
          bumpMap.pixels[i + 2] = v;
          bumpMap.pixels[i + 3] = 255;
        }
      }
      bumpMap.updatePixels();

      describe('A gray sphere lit from the upper left. A grid of round bumps covers its surface.');
    }

    function draw() {
      background(0);

      // Enable orbiting with the mouse.
      orbitControl();

      // Rock the shape so the lighting shifts across it.
      rotateY(sin(millis() * 0.002) * PI * 0.1);

      // Light the sphere from the upper left.
      ambientLight(60);
      pointLight(255, 255, 255, -80, -80, 150);
      noStroke();
      fill(200);

      // Tile the map so it wraps around the sphere without a seam.
      textureWrap(REPEAT);

      // Highlights make the raised areas easier to pick out.
      specularMaterial(255);
      shininess(40);

      // Raise the bumps without changing the geometry.
      bumpTexture(bumpMap, 4);
      sphere(40);
    }
